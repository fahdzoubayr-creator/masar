الاعتبارات والمرجعيات 
روعــي فــي بلــورة هــذا الــرأي الاســتناد إلــى مجموعــة مــن الاعتبــارات والمرجعيــات، هــي علــى التوالــي:  
. الاعتبارات1
 	الوضــع الفــردي والاجتماعــي للأشــخاص فــي وضعيــة إعاقــة المتســم فــي غالبيــة الأحــوال بالهشاشــة، والمعانــاة مــن التمـييـــز 
والإقصــاء، وعــدم الاســتفادة مــن الحقــوق التــي تخولهــا لهــم النصــوص التشــريعية والمرجعيــات الحقوقيــة الوطنيــة 
والدوليــة؛   
	 كــون الإعاقــة مكونــا مــن مكونــات التنــوع الإنســاني، باعتبارهــا نوعــاً مــن الاختــلاف الــذي يتطلــب التفهُّــم، والاعتـــراف 
بــه، علــى كافــة المســتويات النفســـية والحقوقيــة والاجتماعيــة والاقتصاديــة والثقافيــة، واعتمــاد الــلازم مــن الســـياسات 
والمبــادرات لضمــان التمتــع بالحقــوق والقيــام بالواجبــات؛
	 كون الشخص في وضعية إعاقة، إنساناً كامل الإنسانية؛ ذا كرامة؛ وشخصاً حاملاً لحقوق وعليه واجبات؛ ومواطناً 
كامل المواطنة، بما يؤهله للمشاركة الفعلية والفعالة في المجتمع وتنمـيته؛ 
	 كون كل وضعية إعاقة فردية، حالة يتعيـن أخذها بعيـن الاعتبار، والاجتهاد في تنمـية شروط تجاوز وضعها وعوائقها، 
وابتكار الحلول التي تستجيب لحاجاتها، وما تحمله من إمكانات للاستثمار لفائدة الشخص في وضعية إعاقة؛ 
	 اعتبار الحق في التـربية الدامجة، حقا كونيا يشمل جمـيع الأشخاص، ذكورا وإناثا، وأيا كانت اختلافاتهم الاجتماعية 
والاقتصادية واللغوية والديـنية والثقافية، والمجالية، إلخ، بمن في ذلك الأشخاص في وضعية إعاقة؛
	 كــون الارتقــاء بوضــع وحقــوق الأشــخاص فــي وضعيــة إعاقــة، جــزء لا يتجــزأ مــن استـــراتيجيات التنمـــية المســتدامة، 
وخصوصاً ضمن الســـياق الحالي لمراجعة وتجديد النموذج التنموي المغربي، وتحدي مشاركة كل مكونات المجتمع في 
إنجــاح هــذا النمــوذج، أفــراداً ومجتمعــاً محليــاً وجهويــاً أو وطنيــاً؛

	 يشــمل هــذا العــرض التـــربوي لمرحلــة انتقاليــة كلا مــن الأقســام العاديــة، وأقســام الدمــج داخــل المدرســة، والبنيــات أو 
المراكــز المتخصصــة التــي يـــنبغي أن يكــون الإشــراف البيداغوجــي عليهــا مــن اختصــاص وزارة التـــربية الوطنيــة والتكويـــن 
المهنــي والتعليــم العالــي، والتعليمـيـــن العمومـــي والخــاص، وأن تتحــول إلــى مراكــز جهويــة للمــوارد، تثمـيـــنا لمــا راكمتــه مــن 
خبـــرة، تفيــد الأســر والفاعليـــن التـربوييـــن بالمؤسســات التعليمـــية، وتقــدم لهــم، فــي إطــار المواكبــة، كافــة الاستشــارات 
والخبـــرة المطلوبــة بخصــوص تعلــم الأطفــال فــي وضعيــة إعاقــة؛ 
	 يـــنبغي تأمـيـــن حركيــة المتمدرســـين بشــكل مســتمر ومنتظــم بيـــن هــذه البنيــات الثــلاث للتمــدرس، علــى أن يكــون الانتقــال 
من أقسام الدمج إلى الأقسام العادية، أو من البنيات المختصة إلى المؤسسات التعليمـية وفق خطط ملائمة لمختلف 
الوضعيــات، تســتحضر احتياجــات كل وضعيــة، وتؤمــن لهــا الانتقــال الســلس والاندمــاج دون أيــة معيقــات؛ 
	 بخصــوص الحــالات المســتعصية ذات وضعيــة الإعاقــة الحــادة، التــي تفــرض علــى الطفــل فــي وضعيــة إعاقــة، البقــاء فــي 
المنــزل أو فــي فضــاء الاستشــفاء، يتــم تخصيــص حصــص للتمــدرس المنزلــي أو الاستشــفائي. ويمكــن لهــذا الغــرض تكييــف 
حصــصٍ للتعلــم فــي عيـــن المــكان، بمــا فــي ذلــك، التعليــم عــن بعــد بمســاعدة أطــر مرافقــة مختصــة، وبتوفيـــر فضــاء ملائــم؛ 
	 يـــنبغي توفيـــر كل التيســـيرات الضروريــة للتمــدرس والتعلــم، التــي تخــص تهيئــة الفضــاءات وإعــداد الأطــر ذات الخبـــرة 
والكفــاءة، وتوفيـــر المعــدات والتجهيـــزات.

الخطــط، والتنويــع فــي أســاليب العمــل والوســائل والمــوارد والشــراكات، والتقييــم والرصــد اليقــظ لمــا ســتَؤول إليــه مــن نتائــج، 
وللصعوبــات التــي قــد تعتـــرض مســارات هــذا التفعيــل، وللتدابيـــر الاســتدراكية المواكبــة؛ إن مــن شــأن كل ذلــك، فــي تقديـــر 
المجلــس، أن يســهم بفاعليــة ونجاعــة فــي تحقيــق الأهــداف التــي يـــروم هــذا الــرأي بلوغَهــا. بالنســبة إلــى تعليــم الأطفــال فــي 
وضعيــة إعاقــة، وضمــان تمكنهــم، دون تمـييـــز، مــن حقوقهــم الكاملــة فــي هــذا المجــال.
هكــذا، يعتبـــر المجلــس، أن التـــربية الدامجــة، هــي بالفعــل أفــق مفتــوح، لكنــه مرســومُ الأســس والرافعــات والأهــداف، وقابــل 
دومــا للتفعيــل الواقعــي المتــدرج، المتنــوع الأســاليب والمقاربــات، منفتــحٌ علــى التجديــدات والإغنــاءات المســتجدة والمثمــرة. 
هــذه التـــربية، تشــكل لذلــك وبفضــل ذلــك، خارطــةَ طريــق، لتمكيـــن الأشــخاص فــي وضعيــة إعاقــة، مــن حقوقهــم المشــروعة، 
فــي تعليــم منصــف ذي جــودة للجمـــيع، مـــيسَّر الولــوج، نافــع، ودامــج فرديــا واجتماعيــا؛ وفــي مقومــات تـرســـيخ مواطنتهــم 
بفاعليــة وإنتاجيــة وثقــة فــي الــذات؛ وفــي تنمـــية إمكانهــم فــي القيــام بالواجبــات والمســؤوليات، وفــي تأمـيـــن مشــاركتهم فــي 
التنمـــية الشــاملة والمســتدامة للمجتمــع.
إن الارتقــاء بواقــع الأشــخاص فــي وضعيــة إعاقــة عبـــر أفــق التـــربية الدامجــة، يجسِّــد، إذن، مدخــلا ملموســا مــن مداخــل 
مناهضــة الفــوارق علــى أســاس وضعيــة الإعاقــة، وســبيلا مــن ســبل إقــرار وتوطيــد وتوســـيع مجــال العدالــة الاجتماعيــة 
المنشــودة.
