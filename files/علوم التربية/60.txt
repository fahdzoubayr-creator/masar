 حُٔؼ٘٠ حُٞحعغ ٛٞ ٓـٔٞػش حُوٞحػذ حُؼخٓش حُٔـشدس حُظ٢ طٜذف اُ٠ ط٘ظ٤ْ عِٞى حلأكشحد دحخَ حُٔـظٔغ رؾٌَ ِٓضّ ٝحُٔوظشٗش

رخُذُٝش ك٢ أكذ ٓـخلاص حُل٤خس حلاؿظٔخػ٤ش ٝٛ٘خ ٣ٌٕٞ حُوخٕٗٞ ٓذٝٗخ.

 ٓلّٜٞ حُوخػذس حُوخٗٞٗ٤ش 
هً  الخلٌة  أو الوحدة التً  ٌتألف من  مجموعها  القانون، والتً تخاطب سلوك الأفراد فً المجتمع، وتطبق الجزاء على من ٌخالفها. 
 أنواعها: 
 المدونة( التشرٌع) / غٌر المدونة ( الأعراف).  
 الآمرة ( ٌجب على الأفراد احترامها وتتعلق بالحفاظ على النظام العام ولا ٌمكن المساس بها )  / المكملة (ٌجوز للأفراد 
الاتفاق على ما ٌخالفها، ٌضعها المشرع للمساعدة فً المعاملات المشروعة ). 
 الموضوعٌة( تحدد الحقوق، الواجبات والعلاقات... مثل الدستور) / الشكلٌة ( تبٌن الإجراءات الواجب اتباعها للحصول 
على حق أو قصد معاقبة شخص مثل قانون المسطرة الجنائٌة) 
 خصائصها 
 قاعدة اجتماعٌة ( تتجه إلى تنظٌم وتوجٌه شؤون وقضاٌا المجتمع والتأثٌر فً الحٌاة العامة للمواطنٌن ) 
 قاعدةسلوكٌة ( تتجه إلى تنظٌم السلوك الاجتماعً ولا تهتم بالنواٌا والخواطر). 
 قاعدة عامة(تخاطب عامة الناس بمبدأ المساواة ) ومجردة ( لا تعنً شخصا معٌنا أو حالة محددة، بل تطبق متى 
توافرت شروطها ). 
 قاعدة ملزمة( للسلطة كامل القدرة على فرض احترامها بما تتوفر علٌه من موارد القوة والزجر، أو التعوٌض 
المادي أو موارد الشرعٌة والإقناع لتوجٌه الرأي العام ). 
 مصادر  القاعدة القانونٌة 
 
 
 
 
 
 
 
ٌصاحب القاعدة القانونٌة جزاء توقعه السلطة العامة جبرا على 
من ٌخالفها وٌتخذ صورا متعددة: 
جزاء مدنً بإجبار المدٌن على ما التزم به عٌنا، أو دفع 
مقابل كتعوٌض فً حال عدم التنفٌذ. 
جزاء جنائً بتوقٌع عقوبة على المخالف للقانون وقد 
تكون مالٌة ( الغرامة) أو بدنٌة ( الحبس). 
جزاء إداري ٌترتب عن مخالفة الموظف لأحكام القانون 
الإداري، وقد ٌتخذ شكل الاقتطاع من الأجر، فصل من 
العمل، التوبٌخ، الإنذار أو إلغاء لقرار إداري مشوب 
بإساءة استعمال السلطة ( عٌب من حٌث الشكل أو 
المضمون أو هما معا).

الظهٌر الشرٌف

إعطاء الشرعٌة التنفٌذٌة لقانون صادر عن البرلمان إذ بدون صدور هذا الظهٌر فً الجرٌدة الرسمٌة لا ٌمكن للقانون 
المصادق علٌه من لدن البرلمان أن ٌنتقل إلى مرحلة التنفٌذ.

الفصل الخامس:  تقنٌات التراسل الإداري والوثائق الإدارٌة والمهنٌة.  
1- ٓلّٜٞ حُظشحعَ حلإدحس١ ٝأٛٔ٤ظٚ 
 حُٔلّٜٞ٣وقذ رٚ رُي حُظٞفَ حٌُظخر٢ حُز١ ٣لقَ ر٤ٖ حُٔٞظق ٝحلإدحسس، أٝ ر٤ٖ ادحسس  ٝأخشٟ، ً٤لٔخ ًخٕ ٛذكٚ أٝ ٓٞمٞػٚ، 
ٝرُي ك٢ اهخس حكظشحّ هٞحػذ ٙ ٝحُؾشٝه حُؼخٓش ُِظشحعَ. 
 أٛٔ٤ظٚ 
 طغٜ٤َ ػِٔ٤ش حلاطقخٍ ر٤ٖ حُٔ٘ظٔخص حلإدحس٣ش دٕٝ إٔ ٣ٌٕٞ ٛ٘خى كخؿش اُ٠ حُِوخء حُؾخق٢ ر٤ٖ حُٔغئُٝ٤ٖ.  
 طؼظزش ٓقذسح ٛخٓخ ُِٔؼِٞٓخص ٣ٌٖٔ حُشؿٞع اُ٤ٜخ ػ٘ذحُلخؿش.  
 حُشعخُش رحص حُق٤خؿش حُـ٤ذس ٝ حُقخدهش رٔلظٞحٛخ طلون ػلاهخص ه٤زش ر٤ٖ حلأهشحف رحص حُؼلاهش 
 طٔؼَ حُٔشحعلاص حُٔظزخدُش ٓغظ٘ذح هخٗٞٗ٤خ أٓخّ حُونخس ك٢ كخُش كقٍٞ ٓ٘خصػخص ر٤ٖ حلأهشحف.  
 حُٔشحعلاص أًؼش ؽٔٞلا ٖٓ ٝعخثَ حلاطقخٍ حلأخشٟ ٖٓ ك٤غ حُٔؼِٞٓخص ك٤غ ٣ظ٤ق ٌُخطزٜخ حُظؼشك ُِٔٞحم٤غ رؾ٢ء ٖٓ
حُؾٍٔٞ ٝ حُظلق٤َ حُظ٢ هذ لا طظٞكش ك٢ ٝعخثَ حلاطقخٍ حلأخشٟ. 
 حُشعخُش طغخػذ ػِ٠ حُظؼشف ػِ٠ حُؾخـ حُز١ هخّ رظلش٣شٛخ ٝرخُظخُ٢ ٣ٌٖٔ طلذ٣ذ حُٔغئُٝ٤ش ٝ حُ٘ظخثؾ حُٔظشطزش ػِ٤ٜخ. 
 أٗٞحػٚ 
 سعخُش ؽخق٤ش رطخرغادحس١: طٌظغ٢ حُشعخُش حلإدحس٣ش هخرغ حُؾخق٤ش ٓظ٠ ًخٗض ٓٞؿٜش ٖٓ حُٔٞظل٤ٖ اُ٠ سإعخثْٜ حُٔزخؽش٣ٖ 
لأؿَ سكغ طظِْ أٝ هِذ طٞم٤ق أٝ هِذ ٝػ٤وش ٓؼ٤٘ش. ٝطخنغ ٛزٙ حُشعخُش ُنٞحرو ٜٓ٘خ:  ف٤ـش حُظغٔ٤ش ٝحُ٘ذحء ٝحُٔخخهذ. 
ك٤٘زـ٢ ػِ٠ حُٔشعَ طوذ٣ْ حُٔؼِٞٓخص حُؾخق٤ش ٝحُٜٔ٘٤ش ،ٝحٓخ حُٔشعَ اُ٤ٚ ك٤ظْ حلاًظلخء رخُقلش حُٔؼ٘ٞ٣ش ،ًٔخ إٔ ٛزٙ حُشعخُش 
طخظظْ رؼزخسس ارذحء حُظوذ٣ش ٝحلاكظشحّ. 
 سعخُش ادحس٣ش أٓخ حُشعخُش حلإدحس٣ش كٜ٢ حُٔٞؿٜش ٖٓ حلإدحسس اُ٠ حُؼخِٓ٤ٖ رخُٔئعغش لإخزخسْٛ أٝ طٞؿ٤ْٜٜ أٝ حعظلغخسْٛ ػٖ 
أٓش ٓؼ٤ٖ.  ٝحُلشم ر٤ٜ٘خ ٝر٤ٖ حُشعخُش حُؾخق٤ش حلإدحس٣ش ٣ٌٖٔ ك٢ حُٔقذس ٝحُٞؿٜش ًٝزح ك٢ حُٔؼِٞٓخص حُخخفش رٌَ ٗٞع. 
 
2- حُوٞحػذ حُؼخٓش ُِظشحعَ حلإدحس١ 
٣ظْ حُظشحعَ ر٤ٖ حلإدحسس ٝ حُٔٞظل٤ٖ، ٝ ك٤ٔخ ر٤ٖ حلإدحسس ٝ حُٔئعغخص، ٝكن ٗظخّ ده٤ن ٣٘زـ٢ ػِ٠ حُٔٞظق إٔ ٣طِغ ػِ٤ٚ ٝ ٣لظشّ هٞحػذٙ، 
ٝ الا طؼشمض ٓقخُلٚ ٝ كوٞهٚ ُِن٤خع ٝ حلإٛٔخٍ. 
 حكظشحّ حُغِْ أٝ حُظغِغَ حلإدحس١ : ٣٘زـ٢ إٔ طٞؿٚ حُٔشحعلاص دٝٓخ   اُ٠ حُغِطش حٌُِٔلش أٝ حُٔخظقش، ٓغ ؽشه ٓشٝسٛخ ػزش 
حلإدحسحص ٝ حُٔئعغخص حُٞع٤طش، كخلأعظخر أٝ حُٔذ٣ش أٝ أ١ ٓٞظق رٔئعغش طؼِ٤ٔ٤ش، ارح أسحد إٔ ٣زؼغ سعخُش اُ٠ ادحسس ٓشًض٣ش أٝ 
أًخد٣ٔ٤ش، ػِ٤ٚ إٔ ٣شحعَ حُٔقِلش أٝ حُوغْ حُٔؼ٘٢ رؼذ رًش حُغ٤ذ حُٞص٣ش ٝ حلإدحسحص حُٞع٤طش رخُظ٘خصٍ، ر٤٘ٔخ ٣زًش رخُظقخػذ 
حُـٜخص حُظ٢ عظٔش ػزشٛخ حُشعخُش، ٝ كظ٠ ٣ظْ حُظؤًذ ٖٓ فلش حُظغِغَ حلإدحس١ ك٢ حُٔشحعِش، ٣غظلغٖ حلاػظٔخد ػِ٠ ٛ٤ٌَ 
حُٞصحسس حُٔؼٍٔٞ رٚ. ٝ ٖٓ ؿٜش أخشٟ، ٣ٔ٘غ ػِ٠ أ١ ٓئعغش أٝ أ١ ؽخـ حػظزخس١، إٔ ٣شحعَ علخسس رؾٌَ ٓزخؽش (حُٔزًشس 
حُٞصحس٣ش 104 ك٢ 20/04/1982)، ار لارذ ٖٓ ٓشٝس حُشعخُش ػزش حُغِْ حلإدحس١. رَ ُوذ أفزق حُظشحعَ ر٤ٖ ٓذسعظ٤ٖ أٝ ػخٗٞ٣ظ٤ٖ 
ٓؼلا ، ٣خنغ ُٔشٝسٙ ػزش ٓذ٣ش٣ش حُظؼِ٤ْ، طـ٘زخ   ُِخِو، ٝ طلش٣خ   ُِذهش ٝ حُنزو. 
 حلاعــْ حُٔؼ٘ٞ١ طٞؿٚ حُشعخُش اُ٠ حُٔغئٍٝ حُٔؼ٘٢ رقلظٚ حلإدحس٣ش، دٕٝ حُِـٞء اُ٠ حعٔٚ حُؾخق٢، لإٔ حُشعخُش رحص هخرغ 
ادحس١ طزؼغ ٖٓ حُٔؼ٘٢ رخلأٓش رخػظزخسٙ ؽخقخ  ٓؼ٘ٞ٣خ  اُ٠ حُـٜش حُٔخققش رحطٜخ، رـل حُ٘ظش ػٖ أعٔخء حُٔٞظل٤ٖ حُؼخِٓ٤ٖ 
رٜخ. 
 حُلٔذُش ٝ حُزغِٔش حٗغـخٓخ  ٓغ حُٔزخدة حلإعلآ٤ش ك٢ حُظؼخَٓ ٝ حٌُظخرش، ٝ ٓغ حلأكخد٣غ حُ٘زٞ٣ش حُؾش٣لش، كبٕ حُزذء رـ " رغْ 
الله حُشكٖٔ حُشك٤ْ" أٝ "حُلٔذ لله ٝ كذٙ" ٖٓ حلأخلام حلإعلآ٤ش، ٝ ٣غظلغٖ حلاكظلخظ رٜزٙ حُٔ٤ضس ك٢ ؿٔ٤غ حُٔشحعلاص، ٝ إٔ 
طظٞعو أػِ٠ حُٞسهش.  
 ٝمــــٞف حُخـــو  هذ طظؼشك حُشعخُش ُِن٤خع ٝ حلإٛٔخٍ رغزذ سدحءس حُخو أٝ ػذّ ٝمٞكٚ، رل٤غ طظؼزس هشحءس أٝ كْٜ 
ٓنٕٔٞ حُشعخُش. ُزح لارذ ٖٓ اؿخدس حُخو ك٢ حُٔشحعلاص، ٝ هذ ٣ِـؤ ٖٓ أؿَ رُي اُ٠ حٌُظخرش رخ٥ُش طلش٣خ  ُِٞمٞف ٝ حُـلاء. 
 حُـــــشأط ٜٓٔخ ًخٗض حُـٜش حُٔشعِش، ادحس٣ش أٝ ؽخق٤ش، كبٗٚ ٣ظؼ٤ٖ اػزخص حُشأط ك٢ أػِ٠ حُشعخُش ػِ٠ حُ٤ٔ٤ٖ، ( ٝػِ٠ 
حُ٤غخس إ ًخٗض رخُِـش حلأٓخص٣ـ٤ش  أٝ اكذٟ حُِـخص حلأؿ٘ز٤ش ) ٝ ٛٞ ٣ٌٕٞ ك٢ اكذٟ حُقٞسط٤ٖ:

ٝحكذ أٝ ػذس ٓئحؿش٣ٖ ٝ آخ ُلغخد سإعخء حُٔوخٝلاص حُٔز٤٘ش رؼذٙ ُٝٞ ًخٕ حُٔئحؿش لا ٣ضحٍٝ ٜٓ٘ش طذس ػِ٤ٚ سرلخ ، ٝ رُي كظ٠ ُٝٞ ًخٗض 
ٛزٙ حُلخدػش ٗخؿٔش ػٖ كخُش هٞس هخٛشس أٝ ًخٗض أكٞحٍ حُؾـَ هذ طغززض ك٢ ٓلؼٍٞ حُوٞس حُطز٤ؼ٤ش أٝ صحدص ك٢ خطٞسطٚ ، حُِْٜ الا ارح رشٖٛ
